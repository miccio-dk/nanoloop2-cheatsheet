% ---- sequencer -------------------------------------------------------------------
\posterbox[title=THE SEQUENCER, cseq]{name=seq, column=1, span=5, row=5, rowspan=3}
  {\hd{16 steps} in a 4{\texttimes}4 grid; the playing step is darker.\\[2pt]
   \km{\DP}{move cursor}
   \km{\btn{B}}{place note}
   \km{\btn{A}}{cut note / paste on empty step}
   \km{\btn{B}\plus\DP}{edit note (current parameter)}
   \km{\btn{A}\plus\DP}{edit extra parameters}
   \km{\btn{START}}{mono / chord mode (pitch param)}
   \km{\btn{R}\plus\LR}{meta step}
   \km{\btn{SELECT}}{menu 1}
   \note{Default parameter is pitch: \btn{B}\plus\UD\ semitone,
         \btn{B}\plus\LR\ octave.}
   \note{\hd{Meta step:} \btn{R}\plus\Lf/\Rt/\D\ makes a step play only
         every 2nd or 4th pattern round.}}

% ---- channels ----------------------------------------------------------------------
\posterbox[title=CHANNELS, cgrey]{name=ch, column=6, span=3, row=5, rowspan=3}
  {\chn{r} mono PWM/FM, arp chords, pitch env/LFO\\
   \chn{l} mono PWM/FM, same as R\\
   \chn{s} poly PWM/FM: real chords, stereo phasing, filter/mod env\\
   \chn{n} noise: filter + speed, arp on speed
   \note{R/L/S: filtered pulse wave (variable width) or FM
         (mod amount + mod freq).}
   \note{\hd{Startup:} at the logo, pick colours --
         \btn{START} pink/white, \btn{SELECT} pink/black,
         \U\ grey/white.}}

% ---- sync ----------------------------------------------------------------------------
\posterbox[title=SYNC, cgrey]{name=sync, column=9, span=4, row=5, rowspan=3}
  {\hd{S8} click every 2nd step (default; PO, Volca\ldots)\\
   \hd{S16} 1 click per step\\
   \hd{S24} 3 clicks/step -- for MIDI adaptor\\
   \hd{SY1} clicks on left audio ch. (PO slave; not on GBA)\\[2pt]
   \km{\btn{A}\plus\UD}{set sync mode (in menu 2)}
   \km{\btn{B}\plus\btn{SELECT}}{CV sync (slave mode)}
   \km{\btn{B}}{start: pattern-sync}
   \km{\btn{A}}{start: manual / external}
   \note{CV sync: ``c'' under the tempo, 1 pulse per step.
         MIDI clock goes out via the link port (Pocket-to-MIDI cable).}
   \note{\hd{Manual start:} \btn{START} begins at once and sends sync;
         \btn{B}\plus\btn{A} = slave. \hd{S16 auto-revert:} no clock
         for \textasciitilde2 s $\rightarrow$ back to step 1.}}

  {128 banks \texttimes\ 15 slots per channel. Slot 0 is read-only
   (loads an empty pattern). The bank also carries the waveforms --
   switching banks can change the sound at once.\\[2pt]
   \UD\ select channel \textperiodcentered{}
   \LR\ select slot \textperiodcentered{}
   \btn{START}\ bank menu \textperiodcentered{}
   \btn{SELECT}\ back to menu 1\\[2pt]
   \begin{tabular}{@{}ll@{\hspace{4mm}}ll@{}}
     \btn{B}\plus\D & save pattern to slot &
     \btn{B}\plus\U & load pattern from slot\\
     \btn{A}\plus\D & save all 4 (column) &
     \btn{A}\plus\U & load column\\
     \btn{B}\plus\LR & next slot + load &
     \btn{A}\plus\LR & next column + load all\\
     \btn{B}\plus\btn{R}\plus\U & load pattern with tempo &
     \btn{A}\plus\btn{B}\plus\U & load column on 1st step\\
     \btn{A}\plus\btn{B}\plus\D & delete slot &
     \btn{A}\plus\btn{B}\plus\LR & mute / unmute / solo\\
   \end{tabular}\\[2pt]
   \hd{Lock:} \btn{L}\plus\btn{R} locks saving (``X'' shown),
        against accidental overwriting.\\
   \hd{Bank menu:} \btn{A}\plus\btn{B}\plus\U\ copy bank
        \textperiodcentered{} \btn{A}\plus\btn{B}\plus\D\ paste bank
        \textperiodcentered{} \btn{B}\plus\UD\ edit bank name\\
   \hd{Transfer:} \btn{A}\plus\btn{B}\plus\U\ send bank
        \textperiodcentered{} \btn{A}\plus\btn{B}\plus\D\ receive bank
        \textperiodcentered{} on error: \btn{SELECT} cancels,
