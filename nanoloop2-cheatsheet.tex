% ---- sequencer -------------------------------------------------------------------
\posterbox[title=THE SEQUENCER, cseq]{name=seq, column=1, span=4, row=5, rowspan=3}
  {\hd{16 steps} in a 4{\texttimes}4 grid; the playing step is darker.\\[2pt]
   \km{\DP}{move cursor}
   \km{\btn{B}}{place note}
   \km{\btn{A}}{cut note / paste on empty step}
   \km{\btn{B}\plus\DP}{edit note (current parameter)}
   \km{\btn{A}\plus\DP}{edit extra parameters}
   \km{\btn{START}}{mono / chord mode (pitch param)}
   \km{\btn{R}\plus\LR}{meta step}
   \km{\btn{SELECT}}{menu 1}
   \note{Default parameter is pitch: \btn{B}\plus\UD\ semitone,
         \btn{B}\plus\LR\ octave.}
   \note{\hd{Meta step:} \btn{R}\plus\Lf/\Rt/\D\ makes a step play only
         every 2nd or 4th pattern round.}}

% ---- channels ----------------------------------------------------------------------
\posterbox[title=CHANNELS, cgrey]{name=ch, column=5, span=3, row=5, rowspan=3}
  {\km{\cbox{r}}{mono, arp, pitch env/LFO}
   \km{\cbox{l}}{mono, same as \chn{r}}
   \km{\cbox{s}}{poly, real chords, stereo}
   \km{\cbox{n}}{noise, filter/speed, arp}
   \note{\chn{r}/\chn{l}/\chn{s}: filtered pulse wave (variable
         width) or FM (mod amount + freq). \chn{s} adds stereo
         phasing + filter/mod env; \chn{n} arp is on speed.
         On \chn{n}, \btn{L} toggles GB noise 16/8-bit (waveform).}}

% ---- legend --------------------------------------------------------------------------
\posterbox[title=LEGEND -- terms, cgrey,
  fontupper=\sffamily\footnotesize]{name=leg, column=11, span=2, row=5, rowspan=3}
  {\hd{pattern} 16-step loop of notes + per-step settings\\
   \hd{channel} one of 4 voices: \chn{r}, \chn{l}, \chn{s} tonal, \chn{n} noise\\
   \hd{step} one of 16 positions in a pattern\\
   \hd{slot} cell storing one pattern; 15 per channel\\
   \hd{bank} full set of slots; 128 of them\\
   \hd{song} patterns chained to play in order\\
   \hd{file} menu to save / load patterns}

% ---- sync ----------------------------------------------------------------------------
\posterbox[title=SYNC, cgrey]{name=sync, column=8, span=3, row=5, rowspan=3}
  {\km{\cbox{S8}}{click every 2nd step (default)}
   \km{\cbox{S16}}{1 click per step}
   \km{\cbox{S24}}{3 clicks/step (MIDI adaptor)}
   \km{\cbox{SY1}}{left-ch. clicks (PO slave)}
   \vspace{1.2mm}
   \km{\btn{A}\plus\UD}{set mode (in menu 2)}
   \km{\btn{B}\plus\btn{SELECT}}{CV sync (slave)}
   \km{\btn{B}}{start: pattern-sync}
   \km{\btn{A}}{start: manual / external}
   \note{CV sync: ``c'' under the tempo, 1 pulse per step.
         MIDI clock out via the link port (Pocket-to-MIDI cable).}
   \note{\hd{Manual start:} \btn{START} begins at once and sends sync;
         \btn{B}\plus\btn{A} = slave. \hd{S16 auto-revert:} no clock
         for \textasciitilde2 s $\rightarrow$ back to step 1.}}
